MY APPROACH

처음에는 가중치가 있는 그래프라는 것은 알았지만 다익스트라를 떠올리지 못하고 BFS로 접근했다. 방문 처리를 먼저 해버려서 더 짧은 경로가 나중에 발견되어도 갱신되지 않았다.

MY CODE
Queue<Node> queue = new LinkedList<>();
queue.offer(new Node(start, 0));
visited[start] = true;

while (!queue.isEmpty()) {
    Node cur = queue.poll();
    for (Node next : graph.get(cur.v)) {
        if (visited[next.v]) continue;
        visited[next.v] = true;
        dist[next.v] = dist[cur.v] + next.w;
        queue.offer(next);
    }
}
ANSWER CODE
PriorityQueue<Node> pq = new PriorityQueue<>();
pq.offer(new Node(start, 0));
dist[start] = 0;

while (!pq.isEmpty()) {
    Node cur = pq.poll();
    if (cur.w > dist[cur.v]) continue;
    for (Node next : graph.get(cur.v)) {
        int cost = dist[cur.v] + next.w;
        if (cost < dist[next.v]) {
            dist[next.v] = cost;
            pq.offer(new Node(next.v, cost));
        }
    }
}

static void combination(int depth, int start) {
    if (depth == 3) {
        spread();
        return;
    }
    for (int i = start; i < N * M; i++) {
        int r = i / M, c = i % M;
        if (map[r][c] != 0) continue;
        map[r][c] = 1;
        combination(depth + 1, i + 1);
        map[r][c] = 0;
    }
}
